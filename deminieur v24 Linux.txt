void genererBombe(int ligne, int colonne, int nbBombe, int TabSYS[ligne][colonne]){
    
	srand(time(NULL)); //initialiser au temp actuel
    
    
	for (int i = 0; i < nbBombe; i++) {
    	int x = rand() % ligne; //  x aléa
    	int y = rand() % colonne; //  y aléa
    	if (TabSYS[x][y] == -1){ //si aucun bombe 
        	TabSYS[x][y] = 0;
    	}
    	else {
        	i--; //regener une bombe
    	}
	}
}

void caseajda (int ligne,int cologne,int TabSYS[ligne][cologne],char TabA2[ligne][cologne],int life1,int nbTour1,int nbBombe1,int x,int y){
    
	for(int i =0;i<ligne;i++){
                        	//+1 pour regles les pb de coin
    	for(int j =0;j<cologne;j++){
       	 
        	if (i >= 0 && i <= ligne && j >= 0 && j <= cologne && TabSYS[i][j]!=0){
            	int nbBombe=0;
           	 
           	 
                	// debut casse adja en croix
                	if(TabSYS[i-1][j]==0){
                	nbBombe++;     	 
                	}
               	 
                	if(TabSYS[i+1][j]==0){
                	nbBombe++;
                	}
               	 
                	if(TabSYS[i][j+1]==0){
                	nbBombe++;
                	}
               	 
                	if(TabSYS[i][j-1]==0){
                	nbBombe++;
                	}
                	// fin case adja croix et debut coin case adja
                	if(TabSYS[i-1][j+1]==0){
                	nbBombe++;
                	}
               	 
                	if(TabSYS[i+1][j-1]==0){
                	nbBombe++;
                	}
               	 
                	if(TabSYS[i+1][j+1]==0){
                	nbBombe++;
                	}
            	 
                	if(TabSYS[i-1][j-1]==0){
                	nbBombe++;
                	}
                	// fin coin des case adja
                    	if (nbBombe==0){
                        	TabSYS[i][j]=-1;//si pas de bombe met -1 et             	0 au tab char
                    	}
                    	else TabSYS[i][j]=nbBombe;
        	}
    	}
	}
}
